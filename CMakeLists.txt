cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" MC_RELEASE_VERSION)
string(STRIP "${MC_RELEASE_VERSION}" MC_RELEASE_VERSION)
string(REGEX MATCH "^[0-9]+[.][0-9]+[.][0-9]+" MC_NUMERIC_VERSION "${MC_RELEASE_VERSION}")
if(NOT MC_NUMERIC_VERSION)
  message(FATAL_ERROR "VERSION must start with major.minor.patch")
endif()
project(MotionCompensator VERSION ${MC_NUMERIC_VERSION} LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "MotionCompensator requires Windows")
endif()
configure_file(src/release_version.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/release_version.hpp" @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(mc_preferences STATIC src/preferences.cpp)
target_include_directories(mc_preferences PUBLIC src)
if(WIN32)
  target_link_libraries(mc_preferences PUBLIC advapi32)
endif()
add_executable(preferences_tests tests/preferences_tests.cpp)
target_link_libraries(preferences_tests PRIVATE mc_preferences)
add_test(NAME preferences_tests COMMAND preferences_tests)
if(MINGW)
  target_compile_options(mc_preferences PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(preferences_tests PRIVATE -static)
endif()
option(MC_WITH_OPENCV "Use the validated OpenCV subpixel tracking backend" ON)
if(MC_WITH_OPENCV)
  set(OpenCV_STATIC ON)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/opencv/OpenCVConfig.cmake")
    set(OpenCV_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/opencv")
  endif()
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc video)
endif()
if(MC_WITH_OPENCV AND OpenCV_FOUND)
  add_library(mc_tracking STATIC src/tracking.cpp)
  target_include_directories(mc_tracking PUBLIC src)
  target_include_directories(mc_tracking SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(mc_tracking PUBLIC ${OpenCV_LIBS})
  add_executable(tracking_tests tests/tracking_tests.cpp)
  target_link_libraries(tracking_tests PRIVATE mc_tracking)
  add_test(NAME tracking_tests COMMAND tracking_tests)
  if(MINGW)
    target_link_options(tracking_tests PRIVATE -static)
    target_compile_options(mc_tracking PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  message(STATUS "Motion backend: OpenCV ${OpenCV_VERSION}")
else()
  message(WARNING "OpenCV not available: using legacy sparse block flow")
endif()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
add_test(NAME core_tests COMMAND core_tests)
add_executable(dynamic_tests tests/dynamic_tests.cpp)
target_include_directories(dynamic_tests PRIVATE src)
add_test(NAME dynamic_tests COMMAND dynamic_tests)
add_executable(dot_tests tests/dot_tests.cpp)
target_include_directories(dot_tests PRIVATE src)
add_test(NAME dot_tests COMMAND dot_tests)
add_executable(edge_tests tests/edge_tests.cpp)
target_include_directories(edge_tests PRIVATE src)
add_test(NAME edge_tests COMMAND edge_tests)
add_executable(preview_tests tests/preview_tests.cpp src/preview_gdi.cpp)
target_include_directories(preview_tests PRIVATE src)
target_compile_definitions(preview_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(preview_tests PRIVATE mc_preferences gdi32 user32)
add_test(NAME preview_tests COMMAND preview_tests)
if(MINGW)
  target_link_options(preview_tests PRIVATE -static)
endif()
if(WIN32)
  enable_language(RC)
  set(MC_PRESET_RESOURCE "${CMAKE_CURRENT_SOURCE_DIR}/packaging/initial-settings.ini")
  set(MC_LICENSE_RESOURCE "${CMAKE_CURRENT_SOURCE_DIR}/packaging/third-party-licenses.txt")
  set(MC_PROJECT_NOTICES_RESOURCE "${CMAKE_CURRENT_SOURCE_DIR}/packaging/project-notices.txt")
  set(MC_PROJECT_LICENSE_RESOURCE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE.txt")
  set(MC_MANIFEST_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/app.manifest")
  configure_file(packaging/app.manifest.in "${MC_MANIFEST_RESOURCE}" @ONLY)
  configure_file(packaging/resources.rc.in "${CMAKE_CURRENT_BINARY_DIR}/resources.rc" @ONLY)
  set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/resources.rc" APPEND PROPERTY OBJECT_DEPENDS "${MC_PRESET_RESOURCE};${MC_LICENSE_RESOURCE};${MC_PROJECT_NOTICES_RESOURCE};${MC_PROJECT_LICENSE_RESOURCE};${MC_MANIFEST_RESOURCE}")
  add_executable(MotionCompensator WIN32 "${CMAKE_CURRENT_BINARY_DIR}/resources.rc" src/main.cpp src/capture.cpp src/overlay.cpp src/input.cpp src/preview_gdi.cpp)
  target_compile_definitions(MotionCompensator PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  if(TARGET mc_tracking)
    target_compile_definitions(MotionCompensator PRIVATE MC_HAS_OPENCV=1)
    target_link_libraries(MotionCompensator PRIVATE mc_tracking)
  endif()
  target_link_libraries(MotionCompensator PRIVATE mc_preferences d3d11 d3dcompiler dxgi d2d1 dwrite dwmapi runtimeobject windowsapp user32 gdi32 ole32 comctl32 comdlg32 psapi gdiplus uxtheme)
  if(MINGW)
    target_compile_options(MotionCompensator PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(MotionCompensator PRIVATE -municode -static-libgcc -static-libstdc++ -static)
    target_link_options(core_tests PRIVATE -static)
    target_link_options(dynamic_tests PRIVATE -static)
    target_link_options(dot_tests PRIVATE -static)
    target_link_options(edge_tests PRIVATE -static)
  elseif(MSVC)
    target_compile_options(MotionCompensator PRIVATE /W4 /utf-8 /permissive-)
  endif()
endif()
